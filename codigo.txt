#include <stdio.h>
#include <stdlib.h>

typedef struct data DATA;
struct data {
  int dia;
  int mes;
  int ano;
};

typedef struct estudante ESTUDANTE;
struct estudante {
  float id;
  char nome[100];
  DATA nasc;
};

void cadastrar(){
    FILE* arq; // variável que armazena a função de abrir o arquivo
    ESTUDANTE ctt;
    arq = fopen("arquivo.txt","a"); // abre o arquivo, que será chamado arquivo com extensão .txt, com o modo ab, no caso do arquivo não existir, criando o arquivo, e se ele já existir, irá adicionar novos reggistros ao fim do arquivo

    char op;
  
    if(arq == NULL){ // verifica se o arquivo foi aberto corretamente
        printf("Erro ao abrir o arquivo!\n");
    } else{
        do{
          fflush(stdin);
          printf("Digite o id do estudante: ");
          scanf("%f",&ctt.id);

          fflush(stdin);
          printf("Digite o nome do estudante: ");
          scanf("%s",ctt.nome);

          fflush(stdin);
          printf("Digite a data de nascimento do estudante: ");
          scanf("%d %d %d",&ctt.nasc.dia,&ctt.nasc.mes,&ctt.nasc.ano);

          fprintf(arq, "%.0f %s %d %d %d\n", ctt.id, ctt.nome, ctt.nasc.dia, ctt.nasc.mes, ctt.nasc.ano);

          fflush(stdin);
          printf("Deseja continuar(s/n)?\n");
          scanf(" %c",&op);
        } while(op == 's');
        fclose(arq);
    }
}

void pesquisar(){
  FILE* arq; // variável que armazena a função de abrir o arquivo
  ESTUDANTE ctt;
  arq = fopen("arquivo.txt","r"); // abre o arquivo, que será chamado arquivo com extensão .txt, com o modo rb, para ler os registros do arquivo

  int id;

  if(arq == NULL){ // verifica se o arquivo foi aberto corretamente
      printf("Erro ao abrir o arquivo!\n");
  } else{
      printf("Digite o id do aluno a ser consultado:\n");
      scanf("%d",&id);

      while(fscanf(arq, "%f %s %d %d %d", &ctt.id, ctt.nome, &ctt.nasc.dia, &ctt.nasc.mes, &ctt.nasc.ano) != EOF){
        if(id==ctt.id){
          printf("Nome: %s\n",ctt.nome);
          printf("Data de nascimento: %.2d/%.2d/%.2d\n\n",ctt.nasc.dia,ctt.nasc.mes,ctt.nasc.ano);
        }
      }
  }
  fclose(arq);
}

void atualizar(){
    FILE* arq; // variável que armazena a função de abrir o arquivo
    ESTUDANTE ctt;
    int id, encontrado = 0;
    FILE* temp; // arquivo temporário para armazenar os dados atualizados

    arq = fopen("arquivo.txt", "r"); // abre o arquivo original em modo de leitura
    temp = fopen("temp.txt", "w"); // abre um arquivo temporário em modo de escrita

    if(arq == NULL || temp == NULL){ // verifica se os arquivos foram abertos corretamente
        printf("Erro ao abrir o arquivo!\n");
    }
        printf("Digite o ID do estudante que deseja atualizar: ");
        scanf("%d", &id);

        while(fscanf(arq, "%f %s %d %d %d", &ctt.id, ctt.nome, &ctt.nasc.dia, &ctt.nasc.mes, &ctt.nasc.ano) != EOF){
            if(ctt.id == id){
                encontrado = 1;
                printf("Digite o novo nome do estudante: ");
                scanf("%s", ctt.nome);

                printf("Digite a nova data de nascimento do estudante: ");
                scanf("%d %d %d", &ctt.nasc.dia, &ctt.nasc.mes, &ctt.nasc.ano);
            }
            fprintf(temp, "%.0f %s %.2d/%.2d/%.2d\n\n", ctt.id, ctt.nome, ctt.nasc.dia, ctt.nasc.mes, ctt.nasc.ano); // escreve o registro (atualizado ou não) no arquivo temporário
        }

        fclose(arq);
        fclose(temp);

        if(encontrado){
            remove("arquivo.txt"); // remove o arquivo original
            rename("temp.txt", "arquivo.txt"); // renomeia o arquivo temporário para o nome do arquivo original
            printf("Dados atualizados com sucesso!\n");
        } else {
            remove("temp.txt"); // remove o arquivo temporário se o ID não foi encontrado
            printf("Estudante com ID %d não encontrado.\n", id);
        }
}

void excluir(){
    FILE* arq; // variável que armazena a função de abrir o arquivo
    ESTUDANTE ctt;
    int id, encontrado = 0;
    FILE* temp; // arquivo temporário para armazenar os dados atualizados

    arq = fopen("arquivo.txt", "r"); // abre o arquivo original em modo de leitura
    temp = fopen("temp.txt", "w"); // abre um arquivo temporário em modo de escrita

    if(arq == NULL || temp == NULL){ // verifica se os arquivos foram abertos corretamente
        printf("Erro ao abrir o arquivo!\n");
    }
        printf("Digite o ID do estudante que deseja excluir: ");
        scanf("%d", &id);

            if(ctt.id == id){
                encontrado = 1;
                printf("Estudante com id %d excluído.\n",id);
            } else{
                fprintf(temp, "%.0f %s %.2d/%.2d/%.2d\n\n", ctt.id, ctt.nome, ctt.nasc.dia, ctt.nasc.mes, ctt.nasc.ano); // escreve os registros não excluídos no arquivo temporário

        fclose(temp);

            remove("arquivo.txt"); // remove o arquivo original

            printf("Dados atualizados com sucesso!\n");
        } else {
            remove("temp.txt"); // remove o arquivo temporário se o ID não foi encontrado

        }
}

int main(void) {
  int opcao;

  do {
    printf("Escolha uma opção:\n");
    printf("1 - Cadastrar novos estudantes\n");
    printf("2 - Consultar dados de um estudante\n");
    printf("3 - Atualizar informações de um estudante\n");
    printf("4 - Excluir um estudante\n");
    printf("5 - Sair\n\n");
    scanf("%d", &opcao);
    switch (opcao) {
    case 1:
      cadastrar();
      break;
    case 2:
      pesquisar();
      break;
    case 3:
      atualizar();
      break;
    case 4:
      excluir();
      break;
    case 5:
      printf("Obrigado pela visita!\n");
      break;
    default:
      printf("Opção inválida\n");
      break;
    }
  } while (opcao != 5);
  return 0;
}
